🔑 PRIMARY KEY Constraint

-- Single column primary key CREATE TABLE students ( student_id INT PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(100) UNIQUE ); -- Primary key with AUTO_INCREMENT CREATE TABLE employees ( emp_id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(50) NOT NULL, department VARCHAR(30) ); -- Composite primary key (multiple columns) CREATE TABLE order_items ( order_id INT, product_id INT, quantity INT, price DECIMAL(10,2), PRIMARY KEY (order_id, product_id) ); -- Add primary key to existing table ALTER TABLE customers ADD PRIMARY KEY (customer_id);
PRIMARY KEY Explanation:
student_id INT PRIMARY KEY - Student_id column unique-um NOT NULL-um aagum
AUTO_INCREMENT PRIMARY KEY - Automatic-a increment aagura unique ID
PRIMARY KEY (order_id, product_id) - Two columns combination unique irukanum
One table = One primary key only - Multiple primary keys create panna mudiyaadhu

🔗 FOREIGN KEY Constraint

-- Create parent table first CREATE TABLE departments ( dept_id INT PRIMARY KEY, dept_name VARCHAR(50) NOT NULL ); -- Create child table with foreign key CREATE TABLE employees ( emp_id INT PRIMARY KEY, name VARCHAR(50) NOT NULL, department_id INT, FOREIGN KEY (department_id) REFERENCES departments(dept_id) ); -- Foreign key with actions CREATE TABLE orders ( order_id INT PRIMARY KEY, customer_id INT, order_date DATE, FOREIGN KEY (customer_id) REFERENCES customers(customer_id) ON DELETE CASCADE ON UPDATE CASCADE ); -- Add foreign key to existing table ALTER TABLE employees ADD CONSTRAINT fk_department FOREIGN KEY (department_id) REFERENCES departments(dept_id);

❌ Without Foreign Key:

  • Wrong department_id enter pannalam
  • Non-existing department reference pannalam
  • Data inconsistency varum

✅ With Foreign Key:

  • Valid department_id mattum accept aagum
  • Referenced table-la value irukanum
  • Data integrity maintain aagum
FOREIGN KEY Actions:
ON DELETE CASCADE - Parent record delete aana child records um delete aagum
ON DELETE SET NULL - Parent delete aana child-la NULL set aagum
ON UPDATE CASCADE - Parent ID change aana child-layum update aagum
ON DELETE RESTRICT - Child records irundha parent delete panna mudiyaadhu

⭐ UNIQUE Constraint

-- Single column unique CREATE TABLE users ( user_id INT PRIMARY KEY, username VARCHAR(50) UNIQUE, email VARCHAR(100) UNIQUE NOT NULL, phone VARCHAR(15) UNIQUE ); -- Multiple column unique combination CREATE TABLE products ( product_id INT PRIMARY KEY, product_code VARCHAR(20), category_id INT, UNIQUE (product_code, category_id) ); -- Add unique constraint to existing table ALTER TABLE customers ADD CONSTRAINT uk_email UNIQUE (email); -- Multiple unique constraints CREATE TABLE employees ( emp_id INT PRIMARY KEY, employee_code VARCHAR(10) UNIQUE, email VARCHAR(100) UNIQUE, phone VARCHAR(15) UNIQUE );

❌ UNIQUE Constraint Violations:

-- These will fail: INSERT INTO users (user_id, username, email) VALUES (1, 'john_doe', 'john@email.com'); -- This will fail - duplicate username INSERT INTO users (user_id, username, email) VALUES (2, 'john_doe', 'john2@email.com'); -- Error: Duplicate entry 'john_doe' for key 'username'
UNIQUE vs PRIMARY KEY:
UNIQUE: Multiple columns-la apply pannalam, NULL values allow
PRIMARY KEY: One per table, NULL values allow illa
Both: Duplicate values prevent panrum

❗ NOT NULL Constraint

-- NOT NULL constraint during table creation CREATE TABLE customers ( customer_id INT PRIMARY KEY, first_name VARCHAR(50) NOT NULL, last_name VARCHAR(50) NOT NULL, email VARCHAR(100) NOT NULL UNIQUE, phone VARCHAR(15), -- This allows NULL address TEXT ); -- Add NOT NULL to existing column ALTER TABLE customers MODIFY phone VARCHAR(15) NOT NULL; -- Multiple NOT NULL columns CREATE TABLE orders ( order_id INT PRIMARY KEY, customer_id INT NOT NULL, order_date DATE NOT NULL, total_amount DECIMAL(10,2) NOT NULL, status VARCHAR(20) NOT NULL DEFAULT 'Pending' );

❌ NOT NULL Violations:

-- These will fail: INSERT INTO customers (customer_id, first_name, last_name) VALUES (1, 'John', NULL); -- last_name is NOT NULL INSERT INTO customers (customer_id, first_name) VALUES (2, 'Jane'); -- Missing required last_name and email

✅ CHECK Constraint

-- Basic CHECK constraints CREATE TABLE employees ( emp_id INT PRIMARY KEY, name VARCHAR(50) NOT NULL, age INT CHECK (age >= 18 AND age <= 65), salary DECIMAL(10,2) CHECK (salary > 0), gender ENUM('Male', 'Female', 'Other'), email VARCHAR(100) CHECK (email LIKE '%@%.%') ); -- Complex CHECK constraints CREATE TABLE products ( product_id INT PRIMARY KEY, name VARCHAR(100) NOT NULL, price DECIMAL(10,2) CHECK (price > 0), discount DECIMAL(5,2) CHECK (discount >= 0 AND discount <= 100), stock_quantity INT CHECK (stock_quantity >= 0), category VARCHAR(50) CHECK (category IN ('Electronics', 'Clothing', 'Books', 'Sports')) ); -- Add CHECK constraint to existing table ALTER TABLE employees ADD CONSTRAINT chk_salary CHECK (salary BETWEEN 20000 AND 200000);
CHECK Constraint Examples:
age >= 18 AND age <= 65 - Age 18-65 range-la irukanum
salary > 0 - Salary positive value irukanum
discount >= 0 AND discount <= 100 - Discount 0-100% range-la irukanum
email LIKE '%@%.%' - Basic email format check pannum
category IN ('Electronics'...) - Specific values mattum allow pannum

🏷️ DEFAULT Constraint

-- DEFAULT values during table creation CREATE TABLE orders ( order_id INT AUTO_INCREMENT PRIMARY KEY, customer_id INT NOT NULL, order_date DATE DEFAULT (CURDATE()), status VARCHAR(20) DEFAULT 'Pending', total_amount DECIMAL(10,2) DEFAULT 0.00, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP ); -- DEFAULT with expressions CREATE TABLE employees ( emp_id INT PRIMARY KEY, name VARCHAR(50) NOT NULL, joining_date DATE DEFAULT (CURDATE()), department VARCHAR(30) DEFAULT 'General', is_active BOOLEAN DEFAULT TRUE, salary DECIMAL(10,2) DEFAULT 25000.00 ); -- Add DEFAULT to existing column ALTER TABLE products ALTER COLUMN stock_quantity SET DEFAULT 0;

✅ DEFAULT Benefits:

  • Value kudukkala-na automatic-a default value set aagum
  • Data entry time reduce aagum
  • Consistent values ensure pannum
  • NULL values avoid pannum

📝 Quick Reference

  • PRIMARY KEY: Unique + NOT NULL, one per table
  • FOREIGN KEY: References another table's primary key
  • UNIQUE: No duplicate values, multiple per table
  • NOT NULL: Value mandatory, no empty allowed
  • CHECK: Custom validation rules
  • DEFAULT: Automatic value when not provided
