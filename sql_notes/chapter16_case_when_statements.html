🔍 Basic CASE WHEN Examples

-- Simple grade classification using CASE WHEN SELECT student_name, subject, marks, CASE WHEN marks >= 90 THEN 'A+' WHEN marks >= 80 THEN 'A' WHEN marks >= 70 THEN 'B' WHEN marks >= 60 THEN 'C' WHEN marks >= 50 THEN 'D' ELSE 'F' END AS grade, CASE WHEN marks >= 90 THEN 'Excellent' WHEN marks >= 80 THEN 'Very Good' WHEN marks >= 70 THEN 'Good' WHEN marks >= 60 THEN 'Satisfactory' WHEN marks >= 50 THEN 'Pass' ELSE 'Fail' END AS performance_status FROM student_grades ORDER BY student_name, marks DESC;
Basic CASE Explanation:
WHEN marks >= 90 THEN 'A+' - Marks 90 or above-na A+ grade assign pannum
WHEN marks >= 80 THEN 'A' - 80-89 range-la A grade kudukum
ELSE 'F' - Yaaru conditions-um match aagala-na default value
Multiple CASE columns - Oru query-la multiple CASE statements use pannalam

-- Complex CASE with multiple conditions SELECT student_name, subject, marks, attendance_percentage, -- Grade based on marks and attendance CASE WHEN marks >= 90 AND attendance_percentage >= 90 THEN 'A+ (Excellent)' WHEN marks >= 80 AND attendance_percentage >= 80 THEN 'A (Very Good)' WHEN marks >= 70 AND attendance_percentage >= 75 THEN 'B (Good)' WHEN marks >= 60 AND attendance_percentage >= 70 THEN 'C (Satisfactory)' WHEN marks >= 50 AND attendance_percentage >= 65 THEN 'D (Pass)' WHEN marks >= 50 OR attendance_percentage >= 80 THEN 'D (Conditional Pass)' ELSE 'F (Fail)' END AS final_grade, -- Student category classification CASE WHEN marks >= 85 AND attendance_percentage >= 90 THEN 'Star Student' WHEN marks >= 75 AND attendance_percentage >= 85 THEN 'Regular Student' WHEN marks >= 50 AND attendance_percentage >= 70 THEN 'Average Student' WHEN marks < 50 OR attendance_percentage < 70 THEN 'Needs Attention' ELSE 'Review Required' END AS student_category, -- Scholarship eligibility CASE WHEN marks >= 90 AND attendance_percentage >= 95 THEN 'Full Scholarship' WHEN marks >= 85 AND attendance_percentage >= 90 THEN 'Partial Scholarship' WHEN marks >= 80 AND attendance_percentage >= 85 THEN 'Merit Certificate' ELSE 'Not Eligible' END AS scholarship_status FROM student_grades ORDER BY marks DESC, attendance_percentage DESC;

📊 CASE with Aggregation Functions

-- Conditional aggregation using CASE SELECT student_name, -- Count subjects by performance level COUNT(*) AS total_subjects, SUM(CASE WHEN marks >= 90 THEN 1 ELSE 0 END) AS excellent_subjects, SUM(CASE WHEN marks >= 80 AND marks < 90 THEN 1 ELSE 0 END) AS very_good_subjects, SUM(CASE WHEN marks >= 70 AND marks < 80 THEN 1 ELSE 0 END) AS good_subjects, SUM(CASE WHEN marks < 50 THEN 1 ELSE 0 END) AS failed_subjects, -- Average marks calculation with conditional logic ROUND(AVG(marks), 2) AS average_marks, -- Conditional average (exclude failed subjects) ROUND(AVG(CASE WHEN marks >= 50 THEN marks ELSE NULL END), 2) AS average_passing_marks, -- Overall performance status CASE WHEN AVG(marks) >= 85 THEN 'Outstanding' WHEN AVG(marks) >= 75 THEN 'Very Good' WHEN AVG(marks) >= 65 THEN 'Good' WHEN AVG(marks) >= 50 THEN 'Satisfactory' ELSE 'Needs Improvement' END AS overall_performance FROM student_grades GROUP BY student_name ORDER BY average_marks DESC;
Aggregation with CASE Explanation:
SUM(CASE WHEN marks >= 90 THEN 1 ELSE 0 END) - Condition satisfy aana 1 add pannum, illa-na 0 - count madhiri work aagum
AVG(CASE WHEN marks >= 50 THEN marks ELSE NULL END) - Failed subjects exclude panni average calculate pannum
GROUP BY with CASE - Student-wise aggregated results-ku CASE conditions apply pannum
Nested aggregation - Aggregation result-a CASE condition-la use pannalam

🔗 Nested CASE Statements

-- Nested CASE for complex business logic SELECT student_name, subject, marks, attendance_percentage, -- Nested CASE for detailed grading system CASE WHEN marks >= 80 THEN CASE WHEN attendance_percentage >= 95 THEN 'A+ with Perfect Attendance' WHEN attendance_percentage >= 90 THEN 'A+ with Excellent Attendance' WHEN attendance_percentage >= 85 THEN 'A+ with Good Attendance' ELSE 'A+ with Average Attendance' END WHEN marks >= 70 THEN CASE WHEN attendance_percentage >= 90 THEN 'A with Excellent Attendance' WHEN attendance_percentage >= 80 THEN 'A with Good Attendance' ELSE 'A with Average Attendance' END WHEN marks >= 60 THEN CASE WHEN attendance_percentage >= 85 THEN 'B with Good Attendance' WHEN attendance_percentage >= 75 THEN 'B with Average Attendance' ELSE 'B with Poor Attendance' END ELSE CASE WHEN attendance_percentage >= 80 THEN 'Poor Marks but Good Attendance' ELSE 'Needs Immediate Attention' END END AS detailed_assessment, -- Action recommendations using nested CASE CASE WHEN marks < 50 THEN CASE WHEN attendance_percentage < 70 THEN 'Counseling Required + Extra Classes' WHEN attendance_percentage < 80 THEN 'Extra Tutoring Needed' ELSE 'Study Method Review Required' END WHEN marks < 70 THEN CASE WHEN attendance_percentage < 75 THEN 'Improve Attendance + Study Support' ELSE 'Additional Practice Recommended' END ELSE 'Continue Current Approach' END AS recommendation FROM student_grades ORDER BY student_name, subject;

📈 CASE in Different Contexts

-- CASE in WHERE clause for dynamic filtering SELECT student_name, subject, marks, attendance_percentage FROM student_grades WHERE CASE WHEN subject = 'Mathematics' THEN marks >= 70 WHEN subject = 'Physics' THEN marks >= 65 WHEN subject = 'Chemistry' THEN marks >= 60 ELSE marks >= 50 END = TRUE ORDER BY student_name, marks DESC; -- CASE in ORDER BY for custom sorting SELECT student_name, subject, marks, CASE WHEN marks >= 90 THEN 'A+' WHEN marks >= 80 THEN 'A' WHEN marks >= 70 THEN 'B' ELSE 'C' END AS grade FROM student_grades ORDER BY CASE WHEN marks >= 90 THEN 1 WHEN marks >= 80 THEN 2 WHEN marks >= 70 THEN 3 ELSE 4 END, marks DESC, student_name; -- CASE for data transformation and cleanup SELECT student_name, UPPER(CASE WHEN subject = 'Mathematics' THEN 'MATH' WHEN subject = 'Physics' THEN 'PHY' WHEN subject = 'Chemistry' THEN 'CHEM' ELSE 'OTHER' END) AS subject_code, marks, -- Normalize marks to different scale CASE WHEN marks >= 90 THEN 10.0 WHEN marks >= 80 THEN ROUND((marks - 80) / 10.0 + 8.0, 1) WHEN marks >= 70 THEN ROUND((marks - 70) / 10.0 + 7.0, 1) WHEN marks >= 60 THEN ROUND((marks - 60) / 10.0 + 6.0, 1) WHEN marks >= 50 THEN ROUND((marks - 50) / 10.0 + 5.0, 1) ELSE ROUND(marks / 10.0, 1) END AS normalized_score_out_of_10 FROM student_grades ORDER BY student_name, subject;

⚠️ Common Mistakes & Best Practices

Common Mistakes to Avoid:

❌ Missing ELSE clause: ELSE illa-na conditions match aagala records-ku NULL return aagum
❌ Wrong condition order: CASE sequential-a check pannum, specific conditions first venum
❌ Data type mismatch: THEN clauses-la different data types return panna koodadhu
❌ Too many nested CASE: Complex nesting readability reduce pannum
❌ Performance issues: SELECT list-la complex CASE statements performance affect pannalam

Best Practices:

✅ Always include ELSE: Default value kudunga unexpected NULL values avoid panna
✅ Order conditions properly: Most specific conditions first, general conditions last
✅ Consistent data types: All THEN clauses same data type return pannanum
✅ Use meaningful aliases: CASE expressions-ku descriptive column names kudunga
✅ Comment complex logic: Business logic explain panna comments add pannunga
✅ Test all conditions: Edge cases and boundary values test pannunga

📊 Performance Optimization

Optimization Techniques:

⚡ Index usage: CASE conditions-la use aagura columns-ku index create pannunga
⚡ Avoid functions in CASE: CASE conditions-la heavy functions avoid pannunga
⚡ Use computed columns: Repeated CASE logic-ku computed columns consider pannunga
⚡ Limit nesting: Deep nesting bajama multiple simple CASE statements use pannunga
⚡ WHERE before CASE: Data filter pannitu CASE apply pannunga

-- CASE with window functions SELECT student_name, subject, marks, -- Compare with average marks of the subject CASE WHEN marks > AVG(marks) OVER (PARTITION BY subject) THEN 'Above Average' WHEN marks = AVG(marks) OVER (PARTITION BY subject) THEN 'Average' ELSE 'Below Average' END AS performance_vs_average, -- Rank-based classification CASE WHEN ROW_NUMBER() OVER (PARTITION BY subject ORDER BY marks DESC) <= 2 THEN 'Top Performer' WHEN PERCENT_RANK() OVER (PARTITION BY subject ORDER BY marks DESC) <= 0.25 THEN 'High Performer' WHEN PERCENT_RANK() OVER (PARTITION BY subject ORDER BY marks DESC) <= 0.75 THEN 'Average Performer' ELSE 'Needs Improvement' END AS rank_based_category FROM student_grades ORDER BY subject, marks DESC; -- CASE with subqueries for complex business logic SELECT s.student_name, s.subject, s.marks, -- Compare with class topper CASE WHEN s.marks = (SELECT MAX(marks) FROM student_grades sg WHERE sg.subject = s.subject) THEN 'Subject Topper' WHEN s.marks >= (SELECT AVG(marks) FROM student_grades sg WHERE sg.subject = s.subject) * 1.2 THEN 'Excellent Performance' WHEN s.marks >= (SELECT AVG(marks) FROM student_grades sg WHERE sg.subject = s.subject) THEN 'Good Performance' ELSE 'Below Average Performance' END AS detailed_performance_analysis, -- Improvement needed based on subject difficulty CASE WHEN s.subject IN (SELECT subject FROM student_grades GROUP BY subject HAVING AVG(marks) < 70) AND s.marks < 60 THEN 'Urgent Attention Required' WHEN s.subject IN (SELECT subject FROM student_grades GROUP BY subject HAVING AVG(marks) >= 70) AND s.marks < 50 THEN 'Extra Support Needed' ELSE 'On Track' END AS support_requirement FROM student_grades s ORDER BY s.student_name, s.marks DESC; -- Dynamic CASE for flexible reporting SELECT student_name, -- Dynamic subject performance summary MAX(CASE WHEN subject = 'Mathematics' THEN marks END) AS math_marks, MAX(CASE WHEN subject = 'Physics' THEN marks END) AS physics_marks, MAX(CASE WHEN subject = 'Chemistry' THEN marks END) AS chemistry_marks, -- Grade for each subject MAX(CASE WHEN subject = 'Mathematics' THEN CASE WHEN marks >= 90 THEN 'A+' WHEN marks >= 80 THEN 'A' WHEN marks >= 70 THEN 'B' ELSE 'C' END END) AS math_grade, MAX(CASE WHEN subject = 'Physics' THEN CASE WHEN marks >= 90 THEN 'A+' WHEN marks >= 80 THEN 'A' WHEN marks >= 70 THEN 'B' ELSE 'C' END END) AS physics_grade, MAX(CASE WHEN subject = 'Chemistry' THEN CASE WHEN marks >= 90 THEN 'A+' WHEN marks >= 80 THEN 'A' WHEN marks >= 70 THEN 'B' ELSE 'C' END END) AS chemistry_grade, -- Overall performance CASE WHEN AVG(marks) >= 85 THEN 'Outstanding Student' WHEN AVG(marks) >= 75 THEN 'Excellent Student' WHEN AVG(marks) >= 65 THEN 'Good Student' ELSE 'Average Student' END AS overall_rating FROM student_grades GROUP BY student_name ORDER BY AVG(marks) DESC;
